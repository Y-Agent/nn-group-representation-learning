def text_page(width,height,title,note):
    b=io.BytesIO();c=canvas.Canvas(b,pagesize=(width,height));c.setFont('Helvetica-Bold',16);c.drawString(28,height-30,title)
    # ASCII aliases keep the PDF annotation font portable; plot fonts are embedded.
    plain=note.replace('×','x').replace('M=','M = ')
    c.setFont('Helvetica',9)
    import textwrap
    for i,line in enumerate(textwrap.wrap(plain,145)):c.drawString(28,height-48-i*12,line)
    c.save();b.seek(0);return PdfReader(b).pages[0]

def image_page(path):
    im=Image.open(path);w,h=im.size;b=io.BytesIO();c=canvas.Canvas(b,pagesize=(w,h));c.drawImage(ImageReader(im),0,0,w,h);c.save();b.seek(0);return PdfReader(b).pages[0]

    width=1100.;cellw=(width-56-(columns-1)*16)/columns
    ratios=[float(p.mediabox.height)/float(p.mediabox.width) for p in pages]

    out=text_page(width,height,'Figure '+fig+' | Corrected paper reproduction',CAPTIONS[fig])
    for i,p in enumerate(pages):
        pw,ph=float(p.mediabox.width),float(p.mediabox.height);scale=min(cellw/pw,cellh/ph);col,row=i%columns,i//columns
        x=28+col*(cellw+16)+(cellw-pw*scale)/2;y=height-80-row*(cellh+24)-cellh+(cellh-ph*scale)/2
        out.merge_transformed_page(p,Transformation().scale(scale).translate(x,y))
    writer=PdfWriter();writer.add_page(out)
    with (F/f'figure-{int(fig):02d}.pdf').open('wb') as f:writer.write(f)
