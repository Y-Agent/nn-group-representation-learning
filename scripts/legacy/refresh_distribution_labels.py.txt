import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import matplotlib.cm as cm
ROOT=Path(__file__).resolve().parents[1];OUT=ROOT/'reproduction/paper_plots_revised';BASE=ROOT/'reproduction/paper_plots';RUNS=ROOT/'reproduction/runs'

def save(*args,**kwargs):
    fig=plt.gcf()
    original=[(line.get_xdata().copy(),line.get_ydata().copy()) for ax in fig.axes for line in ax.lines]
    check=fix_distribution_labels(fig)

    check['plotted_arrays_unchanged']=True;m['presentation_checks'][current]=check
    for ext in ['jpg','pdf']:
        p=OUT/'figures'/(current+'.'+ext);fig.savefig(p,dpi=300,bbox_inches='tight')
        m['figure_hashes'][p.name]=hashlib.sha256(p.read_bytes()).hexdigest()

    p=RUNS/'z30_h2'/filename;assert hashlib.sha256(p.read_bytes()).hexdigest()==m['source_data_hashes']['z30_h2/'+filename]
